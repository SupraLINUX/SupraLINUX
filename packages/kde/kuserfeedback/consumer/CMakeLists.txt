cmake_minimum_required(VERSION 3.29)
project(kuserfeedback_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Network Widgets)
find_package(KF6UserFeedback 6.30 REQUIRED CONFIG)
if(NOT TARGET KF6::UserFeedbackCore OR NOT TARGET KF6::UserFeedbackWidgets)
  message(FATAL_ERROR "KUserFeedback exported targets are incomplete")
endif()
add_executable(kuserfeedback-consumer main.cpp)
target_link_libraries(kuserfeedback-consumer PRIVATE Qt6::Core KF6::UserFeedbackCore KF6::UserFeedbackWidgets)
