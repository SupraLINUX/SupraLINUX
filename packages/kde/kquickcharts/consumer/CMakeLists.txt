cmake_minimum_required(VERSION 3.29)
project(kquickcharts_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(KF6QuickCharts 6.30 REQUIRED CONFIG)
if(NOT TARGET KF6::QuickCharts)
  message(FATAL_ERROR "KF6QuickCharts must export KF6::QuickCharts")
endif()
add_executable(kquickcharts-consumer main.cpp)
target_link_libraries(kquickcharts-consumer PRIVATE Qt6::Core KF6::QuickCharts)
