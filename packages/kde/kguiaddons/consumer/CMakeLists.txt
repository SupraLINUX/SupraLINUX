cmake_minimum_required(VERSION 3.29)
project(supralinux_kguiaddons_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9.0 REQUIRED COMPONENTS Gui)
find_package(KF6GuiAddons 6.30.0 REQUIRED)
find_package(KF6CoreAddons 6.30.0 REQUIRED)
add_executable(kguiaddons-consumer main.cpp)
target_link_libraries(kguiaddons-consumer PRIVATE Qt6::Gui KF6::GuiAddons KF6::CoreAddons)
