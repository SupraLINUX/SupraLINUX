cmake_minimum_required(VERSION 3.29)
project(prison_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Gui Multimedia)
find_package(KF6Prison 6.30 REQUIRED CONFIG)
if(NOT TARGET KF6::Prison OR NOT TARGET KF6::PrisonScanner)
  message(FATAL_ERROR "Prison exported targets are incomplete")
endif()
add_executable(prison-consumer main.cpp)
target_link_libraries(prison-consumer PRIVATE Qt6::Core KF6::Prison KF6::PrisonScanner)
