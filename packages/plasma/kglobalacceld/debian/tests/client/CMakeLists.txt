cmake_minimum_required(VERSION 3.16)
project(SupraGlobalShortcutsClient LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets DBus)
find_package(Qt6GuiPrivate 6.10 REQUIRED NO_MODULE)
find_package(KF6GlobalAccel REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6JobWidgets REQUIRED)
if(CANDIDATE_CONSUMER)
    find_package(KGlobalAccelD 6.7.5 EXACT REQUIRED)
else()
    find_package(KGlobalAccelD REQUIRED)
endif()
find_package(X11 REQUIRED COMPONENTS Xtst)
add_executable(shortcut-client main.cpp)
target_link_libraries(shortcut-client PRIVATE KF6::GlobalAccel Qt6::Widgets Qt6::DBus X11::X11 X11::Xtst)
add_executable(direct-sdk-client library.cpp)
target_link_libraries(direct-sdk-client PRIVATE K::KGlobalAccelD Qt6::Widgets Qt6::DBus)
