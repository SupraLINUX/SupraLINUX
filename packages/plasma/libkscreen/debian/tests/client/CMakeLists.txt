cmake_minimum_required(VERSION 3.16)
project(supralinux_screen_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
option(CANDIDATE_CONSUMER "Require exact reviewed candidate SDK" OFF)
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus)
if(CANDIDATE_CONSUMER)
    find_package(KF6Screen 6.7.5 EXACT REQUIRED CONFIG)
else()
    find_package(KF6Screen REQUIRED CONFIG)
endif()
add_executable(screen-client main.cpp)
target_compile_options(screen-client PRIVATE -Wall -Wextra -Werror)
target_link_libraries(screen-client PRIVATE KF6::Screen KF6::ScreenDpms Qt6::Core Qt6::Gui Qt6::DBus)
