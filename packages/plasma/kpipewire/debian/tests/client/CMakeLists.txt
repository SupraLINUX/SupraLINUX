cmake_minimum_required(VERSION 3.16)
project(ubuntu-kpipewire-client LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(KPipeWire 6.6 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PipeWire REQUIRED IMPORTED_TARGET libpipewire-0.3)
add_executable(ubuntu-kpipewire-client main.cpp)
target_link_libraries(ubuntu-kpipewire-client K::KPipeWire K::KPipeWireRecord K::KPipeWireDmaBuf Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
add_executable(private-video-source video-source.c)
target_link_libraries(private-video-source PkgConfig::PipeWire)
target_compile_options(private-video-source PRIVATE -Wall -Wextra -Werror)
