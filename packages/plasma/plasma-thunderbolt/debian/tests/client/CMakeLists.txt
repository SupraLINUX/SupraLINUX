cmake_minimum_required(VERSION 3.16)
project(ubuntu-thunderbolt-client LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets DBus Test Qml Quick)
find_package(KF6CoreAddons 6.24 CONFIG REQUIRED)
find_package(KF6DBusAddons 6.24 CONFIG REQUIRED)
find_package(KF6KCMUtils 6.24 CONFIG REQUIRED)
set(SOURCES main.cpp fakeserver.cpp fakemanager.cpp fakedevice.cpp)
qt_add_dbus_adaptor(SOURCES org.freedesktop.bolt1.device.xml fakedevice.h FakeDevice fakedeviceadaptor FakeDeviceAdaptor)
qt_add_dbus_adaptor(SOURCES org.freedesktop.bolt1.manager.xml fakemanager.h FakeManager fakemanageradaptor FakeManagerAdaptor)
add_executable(ubuntu-thunderbolt-client ${SOURCES})
target_include_directories(ubuntu-thunderbolt-client PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(ubuntu-thunderbolt-client Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus Qt6::Test Qt6::Qml Qt6::Quick KF6::CoreAddons KF6::DBusAddons KF6::KCMUtilsQuick)
