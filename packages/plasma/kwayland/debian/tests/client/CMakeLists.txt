cmake_minimum_required(VERSION 3.16)
project(supralinux_kwayland_consumer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
option(BUILD_CLIENT "Build the installed KWayland client" ON)
option(CANDIDATE_CONSUMER "Require the exact reviewed candidate SDKs" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server)
if(CANDIDATE_CONSUMER)
    find_package(PlasmaWaylandProtocols 1.22.0 EXACT REQUIRED CONFIG)
else()
    find_package(PlasmaWaylandProtocols REQUIRED CONFIG)
endif()
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(XML "${PLASMA_WAYLAND_PROTOCOLS_DIR}/plasma-window-management.xml")
add_custom_command(OUTPUT plasma-server.h COMMAND ${WAYLAND_SCANNER} --strict server-header ${XML} plasma-server.h DEPENDS ${XML})
add_custom_command(OUTPUT plasma-code.c COMMAND ${WAYLAND_SCANNER} --strict private-code ${XML} plasma-code.c DEPENDS ${XML})
add_executable(private-server server.c plasma-server.h plasma-code.c)
target_include_directories(private-server PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(private-server PRIVATE -Wall -Wextra -Werror)
target_link_libraries(private-server PRIVATE PkgConfig::WAYLAND_SERVER)
if(BUILD_CLIENT)
    find_package(Qt6 REQUIRED COMPONENTS Core Gui)
    if(CANDIDATE_CONSUMER)
        find_package(KWayland 6.7.5 EXACT REQUIRED CONFIG)
    else()
        find_package(KWayland REQUIRED CONFIG)
    endif()
    add_executable(kwayland-client main.cpp)
    target_compile_options(kwayland-client PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(kwayland-client PRIVATE Plasma::KWaylandClient Qt6::Core Qt6::Gui)
endif()
